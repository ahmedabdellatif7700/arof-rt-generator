foreach(t rng link_stages transmit_receive samples)
  add_executable(test_${t} test_${t}.cpp)
  target_link_libraries(test_${t} PRIVATE arof_core)
  target_compile_definitions(test_${t} PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")

# Python tools (needs numpy and scipy; skipped otherwise). Point AROF_PYTHON at a venv if needed.
set(AROF_PYTHON "" CACHE FILEPATH "Python with numpy and scipy for the tools test")
if(NOT AROF_PYTHON)
  if(EXISTS ${CMAKE_SOURCE_DIR}/.venv/bin/python)
    set(AROF_PYTHON ${CMAKE_SOURCE_DIR}/.venv/bin/python)
  else()
    find_program(AROF_PYTHON NAMES python3)
  endif()
endif()
if(AROF_PYTHON)
  add_test(NAME tools COMMAND ${AROF_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/test_tools.py $<TARGET_FILE:arof_stream>)
  set_tests_properties(tools PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 600)
endif()
